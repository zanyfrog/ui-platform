const refuse = { recovery: 'refuse' as const, timeoutMs: 40 };
let probes = 0;
const originalKill = process.kill;
process.kill = (() => { probes++; throw Object.assign(new Error('injected dead fixture owner'), { code: 'ESRCH' }); }) as typeof process.kill;
process.env.UI_GRAPH_LOCK_TRACE = '0'; delete process.env.UI_GRAPH_LOCK_METRICS;

async function fixture(run: (f: Awaited<ReturnType<typeof setup>>) => Promise<void>) {
  const f = await setup();
  try { await run(f); }
  finally {
    inject();
    assert.ok(path.resolve(f.root).startsWith(path.resolve(os.tmpdir()) + path.sep + 'd6-r1-'));
    await fs.rm(f.root, { recursive: true, force: true });
  }
}
async function setup() {
  const root = await fs.mkdtemp(path.join(os.tmpdir(), 'd6-r1-'));
  const directory = path.join(root, 'one'); await fs.mkdir(directory);
  const files = { 'page.html': '<section data-uip-node="saved-card"></section>\n', 'page.css': '', 'page.visual.json': JSON.stringify({ schemaVersion: 1, artifactId: 'original-page', kind: 'page', name: 'one', displayName: 'One', route: '/one' }), 'page.fixtures.json': '{}', 'page.functions.ts': 'export {};\n' };
  for (const [name, value] of Object.entries(files)) await fs.writeFile(path.join(directory, name), value);
  const location = { root, directory, anchor: 'page.visual.json', paths: Object.keys(files) };
  const identity = await fs.realpath(path.join(directory, location.anchor));
  const lock = path.join(root, '.uib', 'locks', storageKey(identity) + '.lock');
  const journal = path.join(root, '.uib', 'transactions', storageKey(identity), 'pending.json');
  configure(root);
  return { root, directory, files, location, identity, lock, journal, service: new VisualArtifactService(root, 'page') };
}
async function put(file: string, text: string) { await fs.mkdir(path.dirname(file), { recursive: true }); await fs.writeFile(file, text); }
const code = (expected: string) => (error: any) => { assert.equal(error.code, expected); return true; };
async function preserved(f: Awaited<ReturnType<typeof setup>>) {
  for (const [name, value] of Object.entries(f.files)) assert.equal(await fs.readFile(path.join(f.directory, name), 'utf8'), value);
}
test('healthy role, graph and actual Visual read/list preserve bytes/hash and release only own locks', () => fixture(async f => {
  const before = probes;
  const snapshot = await withRuntimeGraphRead(f.root, () => readRoleBundle(f.location, refuse), refuse);
  assert.deepEqual(snapshot.files, f.files);
  assert.equal((await readRoleBundle(f.location)).checksum, snapshot.checksum);
  const a = await f.service.read('one/page.visual.json', refuse);
  assert.equal(a.metadata?.artifactId, 'original-page');
  assert.equal((await f.service.list(refuse))[0].hash, a.hash);
  assert.equal((await f.service.read('one/page.visual.json')).hash, a.hash);
  assert.equal(probes, before); await preserved(f);
  assert.deepEqual(await fs.readdir(path.dirname(f.lock)), []);
  const removed = events.filter(e => e.op === 'rm');
  assert.ok(removed.length > 0); assert.ok(removed.every(e => e.file.endsWith('.lock')));
}));
test('valid foreign/dead owner remains busy without probe, callback, reclaim or guard', () => fixture(async f => {
  await put(f.lock, '7654321'); const before = probes; let calls = 0;
  const started = performance.now();
  await assert.rejects(withArtifactLock(f.root, f.directory, async () => { calls++; }, f.identity, refuse), code('source.busy'));
  assert.ok(performance.now() - started < 1500); assert.equal(calls, 0); assert.equal(probes, before);
  assert.equal(await fs.readFile(f.lock, 'utf8'), '7654321');
  assert.ok(!events.some(e => e.op === 'rm' || e.file.endsWith('.recovery')) || !events.some(e => e.op.startsWith('open') && e.file.endsWith('.recovery')));
  assert.ok(!events.some(e => e.op === 'rm')); await preserved(f);
}));
test('empty, malformed and ambiguous owner publication stays unresolved', () => fixture(async f => {

}));

test('pending, malformed, legacy and missing-anchor journals refuse without reading or replaying bytes', () => fixture(async f => {
  for (const text of ['{bad', '{}', JSON.stringify({ bundle: f.directory, identity: f.identity, original: f.files })]) {
    await put(f.journal, text); configure(f.root);
    await assert.rejects(readRoleBundle(f.location, refuse), code('source.recovery-required'));
    await assert.rejects(f.service.list(refuse), code('source.recovery-required'));
    let calls = 0;
    await assert.rejects(withRoleBundleGraph(f.root, async () => { calls++; }, refuse), code('source.recovery-required'));
    assert.equal(calls, 0); assert.equal(await fs.readFile(f.journal, 'utf8'), text);
    assert.ok(!events.some(e => e.op === 'readFile' && e.file === f.journal));
    assert.ok(!events.some(e => e.op === 'rm' && !e.file.endsWith('.lock'))); await preserved(f);
  }
  await fs.rm(path.join(f.directory, f.location.anchor));
  await assert.rejects(f.service.list(refuse), code('source.recovery-required'));
}));
test('unknown transaction membership and injected status failure are sanitized recovery refusals', () => fixture(async f => {
  await put(path.join(f.root, '.uib', 'transactions', 'unlocatable'), 'opaque');
  await assert.rejects(recoverRoleBundleRoot(f.root, refuse), code('source.recovery-required'));
  await fs.rm(path.join(f.root, '.uib', 'transactions', 'unlocatable'));

test('bounded queue contention and cancellation do not enter or jump a live holder', () => fixture(async f => {
  let release!: () => void, entered!: () => void;
  const ready = new Promise<void>(r => entered = r), hold = new Promise<void>(r => release = r);
  const first = withArtifactLock(f.root, f.directory, async () => { entered(); await hold; }, f.identity, refuse);
  await ready; let calls = 0;
  const abort = new AbortController(); const reason = new Error('fixture cancelled');
  const pending = withArtifactLock(f.root, f.directory, async () => { calls++; }, f.identity, { ...refuse, timeoutMs: 500, signal: abort.signal });
  abort.abort(reason); await assert.rejects(pending, error => error === reason);
  await assert.rejects(withArtifactLock(f.root, f.directory, async () => { calls++; }, f.identity, refuse), code('source.busy'));
  assert.equal(calls, 0); release(); await first;
  await withArtifactLock(f.root, f.directory, async () => { calls++; }, f.identity, refuse); assert.equal(calls, 1);
}));
