import test from'node:test';import assert from'node:assert/strict';import '../explicit-plain/source.test.js';
import{changeAppearance,readAppearance,appearanceTarget,validAppearance}from'../../src/client/visual-artifacts/appearance-css.js';
const source='<section data-uip-node="root" class="uip-layout-item"><p data-uip-node="text">Hello</p><p data-uip-node="other" class="note">Other</p></section>';
const set=(css:string,property:any,value:string)=>changeAppearance(css,source,'text',false,property,value);
test('appearance allows bounded literal values and rejects CSS injection',()=>{for(const [p,values]of Object.entries({'background-color':['#abc','#ffffff','#ffffff00','transparent'],color:['#172033'],'font-size':['16px','1.25rem','0.5em'],'font-weight':['normal','700'],'text-align':['center','start']}))for(const v of values)assert.ok(validAppearance(p as any,v));for(const[p,v]of [['color','red;display:none'],['color','url(x)'],['color','var(--x)'],['font-size','-2px'],['font-size','1px!important'],['font-weight','1000'],['text-align','middle']])assert.equal(validAppearance(p as any,v),false);});
test('owned changes preserve manual stylesheet bytes and other owned nodes exactly',()=>{const manual='/* authored comment */\r\n.note { color: red !important; border: 1px solid black; }\r\n';let css=set(manual,'color','#172033');assert.ok(css.startsWith(manual));css=changeAppearance(css,source,'other',false,'font-weight','700');const other=css.slice(css.indexOf('[data-uip-node="other"]'));const edited=set(css,'color','#ffffff');assert.ok(edited.startsWith(manual));assert.equal(edited.slice(edited.indexOf('[data-uip-node="other"]')),other);assert.equal(readAppearance(edited,'text').color,'#ffffff');const reset=set(edited,'color','');assert.ok(reset.startsWith(manual));assert.ok(!reset.includes('[data-uip-node="text"]'));assert.equal(readAppearance(reset,'other')['font-weight'],'700');});
test('reset removes only one owned declaration and preserves remaining bytes',()=>{let css=set('','color','#172033');css=set(css,'font-size','24px');const next=set(css,'color','');assert.equal(readAppearance(next,'text').color,undefined);assert.equal(readAppearance(next,'text')['font-size'],'24px');assert.equal(set(next,'color',''),next);});
test('manual selected declarations, shorthand, important and shared rules refuse',()=>{for(const css of ['p{color:red}','[data-uip-node="text"]{color:red!important}','*{color:red}','p{all:unset}','p{animation:fade 1s}','p{font:20px serif}'])assert.throws(()=>set(css,css.includes('font:')?'font-size':'color',css.includes('font:')?'20px':'#ffffff'));assert.throws(()=>changeAppearance('',source,'text',false,'color','#fff','p{color:red}'));});
test('unknown CSS shape, imports, media and ambiguous owned areas refuse without normalization',()=>{for(const css of ['@import "x.css";','@media(max-width:600px){p{color:red}}','p > span{color:red}','p{color:red','/* unclosed','/* uip-appearance:start */','/* uip-appearance:end */','/* uip-appearance:start */ /* uip-appearance:start */ /* uip-appearance:end */','/* uip-appearance:start */[data-uip-node="text"]{color:red!important;}/* uip-appearance:end */','/* uip-appearance:start */[data-uip-node="text"]{color:#fff;color:#000;}/* uip-appearance:end */'])assert.throws(()=>set(css,'color','#ffffff'));});
test('editor-owned area must be top-level and unique',()=>{const css=set('','color','#fff');assert.throws(()=>set('p{'+css+'}','color','#000'));assert.throws(()=>set(css+css,'color','#000'));assert.throws(()=>set(css.replace('color: #fff;','border: 1px;'),'color','#000'));});
test('duplicate, manual, unsafe identity and inline-style targets refuse',()=>{for(const html of [source.replace('data-uip-node="other"','data-uip-node="text"'),'<p data-uip-node="text" style="color:red">Hi</p>','<p>Hi</p>','<uip-slot data-uip-node="text"></uip-slot>'])assert.throws(()=>appearanceTarget(html,'text',false));});
test('Page Region and Component boundaries are explicit',()=>{assert.throws(()=>appearanceTarget(source,'text',true));const composed='<main data-uip-node="region" data-uip-region="region-main">'+source+'</main>';assert.equal(appearanceTarget(composed,'text',true).id,'text');assert.throws(()=>appearanceTarget('<div data-uip-node="c" data-uip-component="card">'+source+'</div>','text',false));});

test('comment-like text in quoted CSS values cannot hide cascade conflicts',()=>{assert.throws(()=>set('p{content:"/*";color:red;} .note{content:"*/";}','color','#fff'));assert.throws(()=>set('p{content:"Hello";}','color','#fff'));});

test('ordinary HTML attribute casing cannot hide inline styles or selector conflicts',()=>{assert.throws(()=>appearanceTarget('<p data-uip-node="text" STYLE="color:red">Hi</p>','text',false));assert.throws(()=>changeAppearance('.note{color:red;}', '<p data-uip-node="text" CLASS="note">Hi</p>','text',false,'color','#fff'));assert.throws(()=>appearanceTarget('<p data-uip-node="text" class="one" CLASS="two">Hi</p>','text',false));assert.throws(()=>appearanceTarget('<p data-uip-node="text" class="one" class="two">Hi</p>','text',false));});
