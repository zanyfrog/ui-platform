const fs = require('node:fs');
const path = require('node:path');
const cli = process.argv[1];
if (cli && path.basename(cli) === 'npm-cli.js') {
 const manifest = JSON.parse(fs.readFileSync(path.resolve(path.dirname(cli), '../package.json'), 'utf8'));
 fs.appendFileSync(process.env.STAB0_NPM_TRACE, JSON.stringify({version:manifest.version,cli,args:process.argv.slice(2),cwd:process.cwd()})+'\n');
 if (manifest.version !== '10.8.2') throw new Error('Unexpected npm version in STAB-0');
}
